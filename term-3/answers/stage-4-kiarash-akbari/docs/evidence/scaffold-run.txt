## 4. the scripts, run standalone with PYTHONPATH unset (they carry their own path insert)

$ /usr/bin/python3 scripts/audit_columns.py --only order_approved_at
(cwd: stage-4)
-- witnesses for the column time audit (ROADMAP phase 2, checklist 4.3)
-- column lists read from stage-2/sql/001_create_olist_schema.sql
-- every column starts unverified; a verdict needs an answer from all three

-- === orders.order_approved_at ===
-- orders.order_approved_at: presence before or at order_approved_at
WITH base AS (
  SELECT o.*
  FROM olist.orders AS o
)
SELECT
  COUNT(*)                                  AS rows_total,
  COUNT(order_approved_at)                  AS approved_rows,
  COUNT(order_approved_at)                             AS order_approved_at_filled,
  COUNT(*) FILTER (WHERE order_approved_at IS NULL) AS never_approved
FROM base;

-- orders.order_approved_at: filled share by order_status
WITH base AS (
  SELECT o.*
  FROM olist.orders AS o
)
SELECT order_status,
  COUNT(*)                  AS rows_total,
  COUNT(order_approved_at)           AS order_approved_at_filled,
  ROUND(100.0 * COUNT(order_approved_at) / NULLIF(COUNT(*), 0), 1) AS filled_pct
FROM base
GROUP BY order_status
ORDER BY rows_total DESC;
